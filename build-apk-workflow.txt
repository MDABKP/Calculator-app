name: Build APK
on:
  workflow_dispatch:
  push:
    branches: [main]
permissions:
  contents: write
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - run: npm install
      - run: npx cap add android
      - run: npx capacitor-assets generate --android --iconBackgroundColor '#16171a' --iconBackgroundColorDark '#16171a' --splashBackgroundColor '#16171a' --splashBackgroundColorDark '#16171a'
      - name: Install custom icons
        run: |
          find android/app/src/main/res -name 'ic_launcher*.webp' -delete
          cp -r icons/res/* android/app/src/main/res/
      - name: Tweak colors and orientation
        run: |
          sed -i 's/#3F51B5/#16171A/g; s/#303F9F/#16171A/g' android/app/src/main/res/values/colors.xml || true
          sed -i 's/android:configChanges/android:screenOrientation="portrait" android:configChanges/' android/app/src/main/AndroidManifest.xml || true
      - name: Compute version automatically
        run: |
          # Release v7 was the last manual build (version 1.2). Build number 8 becomes 1.3, 9 becomes 1.4, and so on.
          MINOR=$(( ${{ github.run_number }} - 5 ))
          if [ "$MINOR" -lt 3 ]; then MINOR=3; fi
          VER="1.$MINOR"
          echo "Version $VER build ${{ github.run_number }}"
          sed -i "s/APP_VERSION=\"[0-9.]*\"/APP_VERSION=\"$VER\"/" www/index.html
          grep -q "APP_VERSION=\"$VER\"" www/index.html || { echo "APP_VERSION not found in www/index.html"; exit 1; }
          echo "VER=$VER" >> "$GITHUB_ENV"
      - run: npx cap sync android
      - name: Set version
        run: |
          echo "Version $VER build ${{ github.run_number }}"
          sed -i 's/versionCode [0-9]*/versionCode ${{ github.run_number }}/' android/app/build.gradle
          sed -i "s/versionName \"[^\"]*\"/versionName \"$VER\"/" android/app/build.gradle
      - name: Use permanent signing key
        run: |
          mkdir -p ~/.android
          cp signing/debug.keystore ~/.android/debug.keystore
      - run: chmod +x android/gradlew && cd android && ./gradlew assembleDebug
      - uses: actions/upload-artifact@v4
        with:
          name: calculator-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
      - uses: softprops/action-gh-release@v2
        with:
          tag_name: v${{ github.run_number }}
          name: Calculator ${{ env.VER }}
          files: android/app/build/outputs/apk/debug/app-debug.apk
